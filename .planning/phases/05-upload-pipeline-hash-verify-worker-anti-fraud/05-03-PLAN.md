```ts
// on-device upload queue is MMKV-backed; this is the server-side re-hash queue.
import { Queue } from 'bullmq';
import Redis from 'ioredis';
let _redis: Redis | undefined;
export function getRedisConnection(): Redis {
  if (_redis) return _redis;
  _redis = new Redis(process.env.REDIS_URL ?? 'redis://localhost:6379', { maxRetriesPerRequest: null });
  return _redis;
}
let _queue: Queue | undefined;
export function getQueue(): Queue {
  if (_queue) return _queue;
  _queue = new Queue('verify', {
    connection: getRedisConnection(),
    defaultJobOptions: { attempts: 5, backoff: { type: 'exponential', delay: 5000 }, removeOnComplete: 1000, removeOnFail: 5000 },
  });
  return _queue;
}
export async function enqueueVerify(recordingId: string): Promise<void> {
  await getQueue().add('verify', { recordingId }, { jobId: recordingId }); // jobId = recordingId → a double-enqueue collapses to one job
}
```

## Trust Boundaries

| Boundary                                             | Description                                                                                                       |
| ---------------------------------------------------- | ----------------------------------------------------------------------------------------------------------------- |
| S3 → worker                                          | The worker `GetObject`s recording bytes (read-only) — the byte-fidelity carve-out; no write-back, no re-mux       |
| Redis ↔ worker / API                                | BullMQ job payloads (`{recordingId}` only) cross here — low sensitivity, but Redis must not be publicly reachable |
| SQS message → worker (prod, via Plan 05-05's poller) | a `recordingId` arrives from an external queue — must be validated against the DB before any state flip           |

## STRIDE Threat Register

| Threat ID | Category                   | Component                                                                                                              | Disposition | Mitigation Plan                                                                                                                                                                                                                       |
| --------- | -------------------------- | ---------------------------------------------------------------------------------------------------------------------- | ----------- | ------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------- |
| T-5-03-01 | Tampering / data integrity | `verifyRecording` fed a stale or forged `recordingId` (redelivered SQS msg, double-enqueue, or a row already verified) | mitigate    | Early-return if the row is missing or `qaStatus !== 'uploaded'` — idempotent; `enqueueVerify` uses `jobId = recordingId` so a double-enqueue collapses to one BullMQ job; the cron re-enqueue also uses the `jobId`-collapse.         |
| T-5-03-02 | Information disclosure     | Redis exposed / unauthenticated                                                                                        | mitigate    | Dev: `redis:7-alpine` is `localhost:6379` only (no auth needed on a dev box). Prod (Plan 05-05 Terraform): ElastiCache in-VPC, security-group-scoped to the ECS tasks; never internet-facing. Job payloads carry only `recordingId`s. |
| T-5-03-03 | Elevation of privilege     | the worker writes outbox rows with a request-derived user id                                                           | mitigate    | `verifyRecording` reads `rec.userId` from the DB row (the recording's true owner) and writes the outbox row with that — never a request- or message-supplied user id.                                                                 |
| T-5-03-04 | Denial of Service          | a 4 GB MP4 OOMs the worker on re-hash                                                                                  | mitigate    | `sha256OfS3Object` streams `GetObject().Body` through `crypto.createHash` via `stream/promises.pipeline` — memory-bounded; never `.transformToByteArray()` / Buffer-collects.                                                         |
| T-5-03-05 | Tampering                  | path traversal / SSRF on the S3 key                                                                                    | mitigate    | The keys come from `recordingKeys({ userId: rec.userId, recordingId })` — deterministic, derived from the DB row's own ids, never from external input; `GetObject` is scoped to `RECORDINGS_BUCKET()`.                                |
| T-5-03-06 | Spoofing / Info disclosure | a forged S3-stored checksum being trusted                                                                              | mitigate    | The worker re-hashes the full object bytes itself (the locked design); it never relies on `GetObjectCommandOutput.ChecksumSHA256` (and `s3-client.ts` runs `WHEN_REQUIRED` so none is stored anyway — Pitfall 10).                    |
