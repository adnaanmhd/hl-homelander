package ai.humynlabs.capture.handdetector
imports: android.graphics.{Bitmap, BitmapFactory}; com.facebook.react.bridge.*; com.facebook.react.module.annotations.ReactModule;
         com.google.mediapipe.framework.image.BitmapImageBuilder; com.google.mediapipe.tasks.core.BaseOptions;
         com.google.mediapipe.tasks.vision.core.RunningMode;
         com.google.mediapipe.tasks.vision.handlandmarker.{HandLandmarker, HandLandmarkerOptions}; java.util.concurrent.Executors

class HumynHandDetectorModule(ctx) : ReactContextBaseJavaModule(ctx) {
override fun getName() = "HumynHandDetector" // companion object { const val NAME = "HumynHandDetector" }
private val executor = Executors.newSingleThreadExecutor()
@Volatile private var landmarker: HandLandmarker? = null
private fun getOrCreate(minConf: Float): HandLandmarker = landmarker ?: synchronized(this) {
landmarker ?: HandLandmarker.createFromOptions(reactApplicationContext,
HandLandmarkerOptions.builder()
.setBaseOptions(BaseOptions.builder().setModelAssetPath("hand_landmarker.task").build()) // CPU delegate (default — do NOT setDelegate(GPU))
.setRunningMode(RunningMode.IMAGE).setNumHands(2)
.setMinHandDetectionConfidence(minConf).setMinHandPresenceConfidence(0.5f).setMinTrackingConfidence(0.5f)
.build()).also { landmarker = it }
}
@ReactMethod fun detectHands(path: String, minConfidence: Double, promise: Promise) {
executor.execute { var decoded: Bitmap? = null; var scaled: Bitmap? = null
try {
val opts = BitmapFactory.Options().apply { inPreferredConfig = Bitmap.Config.RGB_565 } // HAND-13
decoded = BitmapFactory.decodeFile(path, opts) ?: throw IllegalArgumentException("decodeFile returned null for $path")
scaled = Bitmap.createScaledBitmap(decoded, 320, 240, true) // HAND-13
val mc = minConfidence.toFloat().coerceIn(0f, 1f)
val result = getOrCreate(mc).detect(BitmapImageBuilder(scaled).build())
promise.resolve(result.landmarks().size) // 0 / 1 / 2
} catch (e: Exception) { promise.reject("HAND_DETECT_FAILED", e) }
finally { scaled?.takeIf { it !== decoded }?.recycle(); decoded?.recycle() } // HAND-13
}
}
@ReactMethod fun cleanup(promise: Promise) { synchronized(this) { landmarker?.close(); landmarker = null }; promise.resolve(null) }
}
