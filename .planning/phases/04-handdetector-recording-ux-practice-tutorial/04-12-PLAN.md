- private val bgExecutor = Executors.newSingleThreadExecutor() // single-thread; serialises detections
- @Volatile private var landmarker: HandLandmarker? = null
- private fun getOrCreate(minConf: Float): HandLandmarker = landmarker ?: synchronized(this) { ... } // the synchronized guards CONSTRUCTION only
- @ReactMethod fun detectHands(path, minConfidence, promise) { bgExecutor.execute { ... getOrCreate(mc).detect(BitmapImageBuilder(scaled).build()) ...; catch(e: Exception) -> promise.reject("HAND_DETECT_FAILED", e); finally -> bitmap recycle } }

- private var soundPool: SoundPool? = null ; private val soundIds = mutableMapOf<String, Int>()
